  关键测试命令

  # 全流程 (一键)
  python run_pipeline.py --ticker AAPL

  # 跳过 RL
  python run_pipeline.py --ticker AAPL --skip-rl

  # Walk-Forward 模式
  python run_pipeline.py --ticker AAPL --walk-forward

  # 自定义配置
  python run_pipeline.py --config my_config.yaml

  # 单步训练
  python train_signal.py --ticker AAPL --model lightgbm
  python train_rl.py --ticker AAPL --algorithm PPO
